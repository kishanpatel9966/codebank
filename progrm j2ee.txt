import java.sql.CallableStatement;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
public class CallableDemo {
private static final String URL =
"jdbc:mysql://localhost:3306/employee_db?useSSL=false&serverTimezone=UTC";
private static final String USER = "root";
private static final String PASS = ""; // default for WAMP
public static void main(String[] args) {
try {

Class.forName("com.mysql.cj.jdbc.Driver");
Connection conn = DriverManager.getConnection(URL, USER, PASS);
System.out.println("Connected to employee_db");
// Calling Stored Procedure
CallableStatement cs = conn.prepareCall("{CALL getEmployees()}");
ResultSet rs = cs.executeQuery();
System.out.println("\n--- Employee Records ---");
while (rs.next()) {
System.out.println("ID: " + rs.getInt("id") +
", Name: " + rs.getString("name") +
", Salary: " + rs.getDouble("salary"));
}
conn.close();
} catch (Exception e) {
e.printStackTrace();
}
}
}

package com.session;
import jakarta.servlet.*;
import jakarta.servlet.http.*;
import java.io.IOException;
import java.io.PrintWriter;
public class CookieServlet extends HttpServlet {
protected void doPost(HttpServletRequest request, HttpServletResponse response)
throws ServletException, IOException {
response.setContentType("text/html");
PrintWriter out = response.getWriter();
String name = request.getParameter("uname");
// Create Persistent Cookie
Cookie ck = new Cookie("username", name);
// Set cookie age (1 day = 24*60*60 seconds)
ck.setMaxAge(24 * 60 * 60);
response.addCookie(ck);
out.println("<h2>Cookie Created Successfully!</h2>");
out.println("<h3>Welcome " + name + "</h3>");
out.println("<a href='CookieServlet'>Click to Read Cookie</a>");
}

protected void doGet(HttpServletRequest request, HttpServletResponse response)
throws ServletException, IOException {
response.setContentType("text/html");
PrintWriter out = response.getWriter();
Cookie[] cookies = request.getCookies();
if (cookies != null) {
for (Cookie c : cookies) {
if (c.getName().equals("username")) {
out.println("<h2>Welcome Back " + c.getValue() + "</h2>");
}
}
} else {
out.println("<h2>No Cookie Found</h2>");
}
}
}
